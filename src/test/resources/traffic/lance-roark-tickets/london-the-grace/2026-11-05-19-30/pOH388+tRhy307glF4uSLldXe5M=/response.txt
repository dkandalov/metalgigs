					}

			$("#ticket_qty").html(ticket_qty + " ticket" + (ticket_qty == 1 ? '' : 's') + " selected" );

			if(ticket_qty == 9992){
				$("#delivery_output").html("Select tickets to display delivery option(s)");
			} else if(ir == 1){
